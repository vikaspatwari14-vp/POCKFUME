const { Pool } = require("pg"); module.exports = async function handler(req, res) { res.setHeader("Cache-Control", "no-store"); if (req.method !== "GET") { return res.status(405).json({ error: "Method not allowed" }); } const token = process.env.ADMIN_API_TOKEN; if (!token || req.headers.authorization !== `Bearer ${token}`) { return res.status(401).json({ error: "Admin authentication failed" }); } const connectionString = process.env.DATABASE_URL || process.env.STORAGE_URL || process.env.POSTGRES_URL; if (!connectionString) { return res.status(500).json({ error: "Database connection variable is missing" }); } const pool = new Pool({ connectionString, ssl: { rejectUnauthorized: false } }); try { const result = await pool.query( "SELECT * FROM orders ORDER BY created_at DESC LIMIT 500" ); return res.status(200).json({ orders: result.rows }); } catch (error) { console.error("Database query failed:", error.message); return res.status(500).json({ error: "Database query failed. Check the database configuration and orders table." }); } finally { await pool.end(); } };
